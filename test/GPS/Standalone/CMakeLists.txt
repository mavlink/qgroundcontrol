# Register a QtTest consumer of the production libraries.
function(qgc_add_gps_library_test name)
    foreach(dependency IN LISTS ARGN)
        if(NOT TARGET ${dependency})
            return()
        endif()
    endforeach()
    add_executable(${name} ${name}.cc)
    set_target_properties(${name} PROPERTIES AUTOMOC ON)
    target_link_libraries(${name} PRIVATE Qt6::Test ${ARGN})
    add_test(NAME ${name} COMMAND ${name})
    set_tests_properties(${name} PROPERTIES LABELS "Unit;GPS;Library" TIMEOUT 60)
endfunction()
qgc_add_gps_library_test(NTRIPLibraryTest QGCGPSNTRIPNetwork)
qgc_add_gps_library_test(PositionLibraryTest QGCGPSPositioning)
qgc_add_gps_library_test(IOLibraryTest QGCIO QGCWire QGCTiming)
