cmake_minimum_required(VERSION 3.25)
project(GPSBoundaryFixture LANGUAGES NONE)
if(NOT QGC_GPS_SOURCE_DIR OR NOT CASE)
    message(FATAL_ERROR "GPS boundary fixture requires QGC_GPS_SOURCE_DIR and CASE")
endif()
include("${QGC_GPS_SOURCE_DIR}/LibraryBoundaries.cmake")
add_library(QGCGPSProbe INTERFACE)
add_library(Intermediate INTERFACE)
add_library(QGCSettings INTERFACE)
if(CASE STREQUAL "valid")
    target_include_directories(Intermediate INTERFACE "${CMAKE_CURRENT_BINARY_DIR}/QGroundControl")
    target_link_libraries(QGCGPSProbe INTERFACE Intermediate)
elseif(CASE STREQUAL "direct")
    target_link_libraries(QGCGPSProbe INTERFACE QGCSettings)
elseif(CASE STREQUAL "transitive")
    target_link_libraries(QGCGPSProbe INTERFACE "$<LINK_ONLY:Intermediate>")
    target_link_libraries(Intermediate INTERFACE QGCSettings)
elseif(CASE STREQUAL "alias")
    add_library(Example::Intermediate ALIAS Intermediate)
    target_link_libraries(QGCGPSProbe INTERFACE Example::Intermediate)
    target_link_libraries(Intermediate INTERFACE QGCSettings)
elseif(CASE STREQUAL "late-target")
    target_link_libraries(QGCGPSProbe INTERFACE ForwardDeclared)
    qgc_check_gps_library_boundaries("${CMAKE_CURRENT_SOURCE_DIR}")
    cmake_language(DEFER CALL qgc_check_gps_library_boundaries "${CMAKE_CURRENT_SOURCE_DIR}")
    add_library(ForwardDeclared INTERFACE)
    target_link_libraries(ForwardDeclared INTERFACE QGCSettings)
    return()
elseif(CASE STREQUAL "header")
    target_include_directories(Intermediate INTERFACE "${QGC_GPS_SOURCE_DIR}/Integration")
    target_link_libraries(QGCGPSProbe INTERFACE Intermediate)
else()
    message(FATAL_ERROR "Unknown boundary fixture case ${CASE}")
endif()
qgc_check_gps_library_boundaries("${CMAKE_CURRENT_SOURCE_DIR}")
