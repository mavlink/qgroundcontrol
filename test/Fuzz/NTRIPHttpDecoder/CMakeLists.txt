cmake_minimum_required(VERSION 3.25)
project(QGCNTRIPHttpDecoderFuzz LANGUAGES CXX)
enable_testing()

option(QGC_NTRIP_LIBFUZZER "Build the NTRIP HTTP decoder with libFuzzer, ASan, and UBSan" OFF)
find_package(Qt6 6.8 REQUIRED COMPONENTS Core Network)

set(_root "${CMAKE_CURRENT_SOURCE_DIR}/../../..")
set(QGC_GPS_COMPONENTS_ENABLED NTRIPHttp)
add_subdirectory("${_root}/src/GPS/NTRIP" NTRIP)

add_executable(ntrip_http_decoder_smoke NTRIPHttpDecoderFuzzer.cc NTRIPHttpDecoderSmoke.cc)
target_link_libraries(ntrip_http_decoder_smoke PRIVATE QGC::GPSNTRIPHttp)
add_test(NAME NTRIPHttpDecoderFuzzSmoke COMMAND ntrip_http_decoder_smoke)
set_tests_properties(NTRIPHttpDecoderFuzzSmoke PROPERTIES LABELS "Unit;GPS;Fuzz" TIMEOUT 60)

if(QGC_NTRIP_LIBFUZZER)
    if(NOT CMAKE_CXX_COMPILER_ID MATCHES "Clang" OR MSVC)
        message(FATAL_ERROR "QGC_NTRIP_LIBFUZZER requires a Clang driver with libFuzzer and sanitizer runtimes")
    endif()
    add_executable(ntrip_http_decoder_fuzzer NTRIPHttpDecoderFuzzer.cc)
    target_link_libraries(ntrip_http_decoder_fuzzer PRIVATE QGC::GPSNTRIPHttp)
    foreach(target QGCGPSNTRIPHttp ntrip_http_decoder_smoke ntrip_http_decoder_fuzzer)
        target_compile_options(${target} PRIVATE -fsanitize=fuzzer-no-link,address,undefined -fno-omit-frame-pointer)
        target_link_options(${target} PRIVATE -fsanitize=address,undefined)
    endforeach()
    target_link_options(ntrip_http_decoder_fuzzer PRIVATE -fsanitize=fuzzer)
endif()
